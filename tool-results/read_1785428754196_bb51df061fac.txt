   264→// ---------- Create dialog ----------
   265→function ProductFormDialog({
   266→  open,
   267→  onOpenChange,
   268→  categories,
   269→  storeId,
   270→}: {
   271→  open: boolean
   272→  onOpenChange: (b: boolean) => void
   273→  categories: Category[]
   274→  storeId: string
   275→}) {
   276→  const qc = useQueryClient()
   277→  const [form, setForm] = useState({
   278→    title: '',
   279→    subtitle: '',
   280→    description: '',
   281→    thumbnail: '',
   282→    price: '',
   283→    compareAt: '',
   284→    sku: '',
   285→    inventory: '0',
   286→    weightGrams: '',
   287→    origin: '',
   288→    isHandmade: false,
   289→    status: 'published',
   290→    categoryId: '',
   291→    variants: [] as VariantForm[],
   292→  })
   293→
   294→  const createMut = useMutation({
   295→    mutationFn: async () => {
   296→      const res = await fetch('/api/products', {
   297→        method: 'POST',
   298→        headers: { 'Content-Type': 'application/json' },
   299→        body: JSON.stringify({
   300→          storeId,
   301→          ...form,
   302→          price: Number(form.price),
   303→          compareAt: form.compareAt ? Number(form.compareAt) : null,
   304→          inventory: Number(form.inventory),
   305→          weightGrams: form.weightGrams ? Number(form.weightGrams) : null,
   306→          categoryId: form.categoryId || null,
   307→          variants: form.variants.filter(v => v.title.trim()).map(v => ({
   308→            title: v.title,
   309→            sku: v.sku || null,
   310→            price: v.price ? Number(v.price) : null,
   311→            inventory: Number(v.inventory) || 0,
   312→            attributes: v.attributes,
   313→          })),
   314→        }),
   315→      })
   316→      if (!res.ok) throw new Error('Failed')
   317→      return res.json()
   318→    },
   319→    onSuccess: () => {
   320→      qc.invalidateQueries({ queryKey: ['products'] })
   321→      qc.invalidateQueries({ queryKey: ['stats'] })
   322→      toast.success('Product created')
   323→      onOpenChange(false)
   324→    },
   325→    onError: () => toast.error('Failed to create product'),
   326→  })
   327→
   328→  return (
   329→    <Dialog open={open} onOpenChange={onOpenChange}>
   330→      <DialogContent className="max-w-lg max-h-[92vh] overflow-y-auto nice-scroll">
   331→        <DialogHeader>
   332→          <DialogTitle>New product</DialogTitle>
   333→          <DialogDescription>Add a new product to your catalog.</DialogDescription>
   334→        </DialogHeader>
   335→
   336→        <ProductForm form={form} setForm={setForm} categories={categories} />
   337→
   338→        <DialogFooter>
   339→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   340→          <Button
   341→            onClick={() => createMut.mutate()}
   342→            disabled={!form.title || !form.description || !form.price || createMut.isPending}
   343→          >
   344→            {createMut.isPending ? 'Creating…' : 'Create product'}
   345→          </Button>
   346→        </DialogFooter>
   347→      </DialogContent>
   348→    </Dialog>
   349→  )
   350→}

   352→// ---------- Edit sheet ----------
   353→function ProductFormSheet({
   354→  product,
   355→  open,
   356→  onOpenChange,
   357→  categories,
   358→}: {
   359→  product: Product
   360→  open: boolean
   361→  onOpenChange: (b: boolean) => void
   362→  categories: Category[]
   363→}) {
   364→  const qc = useQueryClient()
   365→  const [form, setForm] = useState({
   366→    title: product.title,
   367→    subtitle: product.subtitle || '',
   368→    description: product.description,
   369→    thumbnail: product.thumbnail || '',
   370→    price: String(product.price / 100),
   371→    compareAt: product.compareAt ? String(product.compareAt / 100) : '',
   372→    sku: product.sku || '',
   373→    inventory: String(product.inventory),
   374→    weightGrams: product.weightGrams ? String(product.weightGrams) : '',
   375→    origin: product.origin || '',
   376→    isHandmade: product.isHandmade,
   377→    status: product.status,
   378→    categoryId: product.categoryId || '',
   379→    variants: (product.variants ?? []).map(v => ({
   380→      id: v.id,
   381→      title: v.title,
   382→      sku: v.sku || '',
   383→      price: v.price ? String(v.price / 100) : '',
   384→      inventory: String(v.inventory),
   385→      attributes: v.attributes || {},
   386→    })) as VariantForm[],
   387→  })
   388→
   389→  // Fetch fresh product with variants when sheet opens (in case list view didn't include them)
   390→  useQuery({
   391→    queryKey: ['product', product.id, open],
   392→    queryFn: async () => {
   393→      const res = await fetch(`/api/products/${product.id}`)
   394→      if (!res.ok) return null
   395→      const data = await res.json()
   396→      return data.product as Product & { variants: VariantForm[] }
   397→    },
   398→    enabled: open,
   399→    staleTime: 0,
   400→    refetchOnMount: true,
   401→  })
   402→
   403→  const updateMut = useMutation({
   404→    mutationFn: async () => {
   405→      const res = await fetch(`/api/products/${product.id}`, {
   406→        method: 'PUT',
   407→        headers: { 'Content-Type': 'application/json' },
   408→        body: JSON.stringify({
   409→          ...form,
   410→          price: Number(form.price),
   411→          compareAt: form.compareAt ? Number(form.compareAt) : null,
   412→          inventory: Number(form.inventory),
   413→          weightGrams: form.weightGrams ? Number(form.weightGrams) : null,
   414→          categoryId: form.categoryId || null,
   415→          variants: form.variants.map(v => ({
   416→            ...(v.id ? { id: v.id } : {}),
   417→            title: v.title,
   418→            sku: v.sku || null,
   419→            price: v.price ? Number(v.price) : null,
   420→            inventory: Number(v.inventory) || 0,
   421→            attributes: v.attributes,
   422→            ...(v._destroy ? { _destroy: true } : {}),
   423→          })),
   424→        }),
   425→      })
   426→      if (!res.ok) throw new Error('Failed')
   427→      return res.json()
   428→    },
   429→    onSuccess: () => {
   430→      qc.invalidateQueries({ queryKey: ['products'] })
   431→      qc.invalidateQueries({ queryKey: ['stats'] })
   432→      toast.success('Product updated')
   433→      onOpenChange(false)
   434→    },
   435→    onError: () => toast.error('Failed to update product'),
   436→  })
   437→
   438→  return (
   439→    <Sheet open={open} onOpenChange={onOpenChange}>
   440→      <SheetContent side="right" className="w-full sm:max-w-xl overflow-y-auto nice-scroll">
   441→        <SheetHeader>
   442→          <SheetTitle>Edit product</SheetTitle>
   443→          <SheetDescription>{product.title}</SheetDescription>
   444→        </SheetHeader>
   445→
   446→        <div className="px-1 py-4">
   447→          <ProductForm form={form} setForm={setForm} categories={categories} />
   448→        </div>
   449→
   450→        <SheetFooter>
   451→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   452→          <Button
   453→            onClick={() => updateMut.mutate()}
   454→            disabled={!form.title || !form.description || !form.price || updateMut.isPending}
   455→          >
   456→            {updateMut.isPending ? 'Saving…' : 'Save changes'}
   457→          </Button>
   458→        </SheetFooter>
   459→      </SheetContent>
   460→    </Sheet>
   461→  )
   462→}

   550→        <Select value={form.categoryId as string} onValueChange={(v) => set('categoryId', v)}>
   551→          <SelectTrigger>
   552→            <SelectValue placeholder="Select category" />
   553→          </SelectTrigger>
   554→          <SelectContent>
   555→            {categories.map((c) => (
   556→              <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   557→            ))}
   558→          </SelectContent>
   559→        </Select>
   560→      </div>
   561→      <div className="space-y-1.5">
   562→        <Label>Status</Label>
   563→        <Select value={form.status as string} onValueChange={(v) => set('status', v)}>
   564→          <SelectTrigger>
   565→            <SelectValue />
   566→          </SelectTrigger>
   567→          <SelectContent>
   568→            <SelectItem value="published">Published</SelectItem>
   569→            <SelectItem value="draft">Draft</SelectItem>
   570→          </SelectContent>
   571→        </Select>
   572→      </div>
   573→      <div className="flex items-center justify-between rounded-lg border p-3">
   574→        <div>
   575→          <Label className="cursor-pointer">Handmade</Label>

   577→        </div>
   578→        <Switch checked={form.isHandmade as boolean} onCheckedChange={(v) => set('isHandmade', v)} />
   579→      </div>
   580→
   581→      <VariantsEditor variants={variants} onChange={setVariants} />
   582→    </div>

   586→// ---------- Variants editor ----------
   587→function VariantsEditor({
   588→  variants,
   589→  onChange,
   590→}: {
   591→  variants: VariantForm[]
   592→  onChange: (v: VariantForm[]) => void
   593→}) {
   594→  const visibleVariants = variants.filter((v) => !v._destroy)
   595→
   596→  const addVariant = () => {
   597→    onChange([
   598→      ...variants,
   599→      { title: '', sku: '', price: '', inventory: '0', attributes: {} },
   600→    ])
   601→  }
   602→
   603→  const updateVariant = (idx: number, patch: Partial<VariantForm>) => {
   604→    onChange(variants.map((v, i) => (i === idx ? { ...v, ...patch } : v)))
   605→  }
   606→
   607→  const removeVariant = (idx: number) => {
   608→    const v = variants[idx]
   609→    if (v.id) {
   610→      onChange(variants.map((vv, i) => (i === idx ? { ...vv, _destroy: true } : vv)))
   611→    } else {
   612→      onChange(variants.filter((_, i) => i !== idx))
   613→    }
   614→  }
   615→
   616→  return (
   617→    <div className="space-y-2 rounded-lg border p-3 bg-secondary/20">
   618→      <div className="flex items-center justify-between">
   619→        <div>
   620→          <Label className="text-sm font-semibold">Variants</Label>
   621→          <p className="text-[11px] text-muted-foreground">
   622→            Optional — for size, color, or weight options. If empty, the product is sold as-is.
   623→          </p>
   624→        </div>
   625→        <Button type="button" size="sm" variant="outline" onClick={addVariant}>
   626→          <Plus className="h-3.5 w-3.5 mr-1" />
   627→          Add variant
   628→        </Button>
   629→      </div>
   630→      {visibleVariants.length === 0 ? (
   631→        <p className="text-xs text-muted-foreground/70 italic py-2">
   632→          No variants. Product uses the base price and inventory shown above.
   633→        </p>
   634→      ) : (
   635→        <div className="space-y-2">
   636→          {visibleVariants.map((v, idx) => (
   637→            <div key={idx} className="rounded-md border bg-background p-2.5 space-y-2">
   638→              <div className="flex items-start gap-2">
   639→                <GripVertical className="h-4 w-4 text-muted-foreground/40 mt-2 shrink-0" />
   640→                <div className="flex-1 grid grid-cols-2 gap-2">
   641→                  <Input
   642→                    placeholder="Variant label (e.g. Red / Large, 250g, Size 8)"
   643→                    value={v.title}
   644→                    onChange={(e) => updateVariant(idx, { title: e.target.value })}
   645→                    className="text-sm"
   646→                  />
   647→                  <Input
   648→                    placeholder="SKU (optional)"
   649→                    value={v.sku}
   650→                    onChange={(e) => updateVariant(idx, { sku: e.target.value })}
   651→                    className="text-sm font-mono"
   652→                  />
   653→                </div>
   654→                <Button
   655→                  type="button"
   656→                  size="icon"
   657→                  variant="ghost"
   658→                  className="h-7 w-7 text-destructive hover:text-destructive shrink-0"
   659→                  onClick={() => removeVariant(idx)}
   660→                >
   661→                  <X className="h-3.5 w-3.5" />
   662→                </Button>
   663→              </div>
   664→              <div className="grid grid-cols-2 gap-2 pl-6">
   665→                <div className="flex items-center gap-1.5">
   666→                  <span className="text-[10px] uppercase tracking-wide text-muted-foreground w-12 shrink-0">NPR</span>
   667→                  <Input
   668→                    type="number"
   669→                    placeholder="use base"
   670→                    value={v.price}
   671→                    onChange={(e) => updateVariant(idx, { price: e.target.value })}
   672→                    className="text-sm h-8"
   673→                  />
   674→                </div>
   675→                <div className="flex items-center gap-1.5">
   676→                  <span className="text-[10px] uppercase tracking-wide text-muted-foreground w-12 shrink-0">Stock</span>
   677→                  <Input
   678→                    type="number"
   679→                    placeholder="0"
   680→                    value={v.inventory}
   681→                    onChange={(e) => updateVariant(idx, { inventory: e.target.value })}
   682→                    className="text-sm h-8"
   683→                  />
   684→                </div>
   685→              </div>
   686→              <VariantAttributesEditor
   687→                attrs={v.attributes}
   688→                onChange={(attrs) => updateVariant(idx, { attributes: attrs })}
   689→              />
   690→            </div>
   691→          ))}
   692→        </div>
   693→      )}
   694→    </div>
   695→  )
   696→}
   697→
   698→function VariantAttributesEditor({
   699→  attrs,
   700→  onChange,
   701→}: {
   702→  attrs: Record<string, string>
   703→  onChange: (a: Record<string, string>) => void
   704→}) {
   705→  const entries = Object.entries(attrs)
   706→  const [newKey, setNewKey] = useState('')
   707→  const [newValue, setNewValue] = useState('')
   708→
   709→  const add = () => {
   710→    if (!newKey.trim()) return
   711→    onChange({ ...attrs, [newKey.trim()]: newValue.trim() })
   712→    setNewKey('')
   713→    setNewValue('')
   714→  }
   715→  const remove = (k: string) => {
   716→    const next = { ...attrs }
   717→    delete next[k]
   718→    onChange(next)
   719→  }
   720→
   721→  return (
   722→    <div className="pl-6 space-y-1.5">
   723→      {entries.length > 0 && (
   724→        <div className="flex flex-wrap gap-1">
   725→          {entries.map(([k, val]) => (
   726→            <span
   727→              key={k}
   728→              className="inline-flex items-center gap-1 text-[11px] bg-primary/10 text-primary px-1.5 py-0.5 rounded"
   729→            >
   730→              <span className="font-medium">{k}:</span> {val}
   731→              <button
   732→                type="button"
   733→                onClick={() => remove(k)}
   734→                className="hover:bg-primary/20 rounded p-0.5"
   735→              >
   736→                <X className="h-2.5 w-2.5" />
   737→              </button>
   738→            </span>
   739→          ))}
   740→        </div>
   741→      )}
   742→      <div className="flex gap-1.5">
   743→        <Input
   744→          placeholder="attribute (e.g. color, size, weight)"
   745→          value={newKey}
   746→          onChange={(e) => setNewKey(e.target.value)}
   747→          className="text-xs h-7"
   748→        />
   749→        <Input
   750→          placeholder="value (e.g. Red, L, 250g)"
   751→          value={newValue}
   752→          onChange={(e) => setNewValue(e.target.value)}
   753→          className="text-xs h-7"
   754→        />
   755→        <Button type="button" size="sm" variant="ghost" className="h-7" onClick={add} disabled={!newKey.trim()}>
   756→          <Plus className="h-3 w-3" />
   757→        </Button>
   758→      </div>
   759→    </div>
   760→  )
   761→}
